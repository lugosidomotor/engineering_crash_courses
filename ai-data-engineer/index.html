import pandas as pd
df = pd.DataFrame({"age":[25,35,45],"income":[50k,75k,90k],"purchases":[5,12,3]})
df["income_per_purchase"] = df.income / df.purchases
df["is_active"] = (df.last_days < 30).astype(int)
df["age_bucket"] = pd.cut(df.age, bins=[0,30,40,50,100])

dbt: SQL transzformációk dbt

dbt: SQL transzformációk version control-lal.

[s11]
# models/staging/stg_purchases.sql
WITH purchases AS (
    SELECT * FROM {{ source('raw','purchases') }}
    WHERE amount > 0
)
SELECT user_id,
    COUNT(*) AS purchase_count,
    AVG(amount) AS avg_order,
    SUM(amount) AS total_spend
FROM purchases GROUP BY user_id
Output:
